# Production PowerShell Automation Pack · Emmanuel Kgaphola
# Automates haulier profile creation, RBAC group mapping, and Odoo ticket audit log
param (
    [Parameter(Mandatory = $true)][string]$UserEmail,
    [Parameter(Mandatory = $true)][string]$SiteCode,
    [ValidateSet('Driver', 'Operator', 'Dispatcher')][string]$Role = 'Driver'
)

$ErrorActionPreference = 'Stop'
Write-Host "[INIT] Provisioning $Role profile for $UserEmail at site $SiteCode..." -ForegroundColor Cyan

$roleGroups = @{
    'Driver'     = "ACL-Site-$SiteCode-DriverPortal"
    'Operator'   = "ACL-Site-$SiteCode-WeighbridgeOps"
    'Dispatcher' = "ACL-Site-$SiteCode-DispatchQueue"
}

try {
    $adUser = Get-AzureADUser -ObjectId $UserEmail
    $targetGroup = Get-AzureADGroup -SearchString $roleGroups[$Role]
    Add-AzureADGroupMember -ObjectId $targetGroup.ObjectId -RefObjectId $adUser.ObjectId
    Write-Host "[SUCCESS] Added $UserEmail to $($targetGroup.DisplayName)" -ForegroundColor Green
} catch {
    Write-Error "[FAIL] Provisioning aborted: $_"
}
